import json, os, subprocess

TESTS = [
    ('Т.64: вихід без «Меню»', 'Т.64: вихід на стартовий екран без кліку «Меню» — авто-бій зупиняє battle.close'),

MUTATIONS = [
    ('у `close` (`src/ui/battle.ts`) прибрати `deps.stopAuto()`', 'src/ui/battle.ts',
     '    deps.stopAuto(); // П13.3: авто-бій міг тривати — таймер і стан драйвера скидаємо\n', ''),
    ('у `main.ts` залежність `stopAuto: () => autoBattle.stop()` замінити на `stopAuto: () => undefined`', 'src/main.ts',
     '  stopAuto: () => autoBattle.stop(),\n', '  stopAuto: () => undefined,\n'),
    ('у `exitToStart` (`src/main.ts`) прибрати `battle.close()`', 'src/main.ts',
     '  if (ui.mission !== null) battle.close();\n  geoAnim.cancel();\n  partyOnScreen = false;', '  geoAnim.cancel();\n  partyOnScreen = false;'),

     "): void {\n  battle.close();\n  geoAnim.cancel(); // П6.9", "): void {\n  geoAnim.cancel(); // П6.9"),

def run():
    out = '/tmp/t64/vitest.json'
    if os.path.exists(out):
        os.remove(out)
    subprocess.run(['npx', 'vitest', 'run', 'src/main.ui.test.ts', 'src/ui/battle.test.ts',
                    '--reporter=json', '--outputFile=' + out], capture_output=True, text=True)
    with open(out, encoding='utf-8') as f:
        data = json.load(f)
    status = {}
    for file in data['testResults']:
        for t in file['assertionResults']:
            status[t['title']] = t['status']
    cells = []
    for _, title in TESTS:
        assert title in status, title
        cells.append('зелений' if status[title] == 'passed' else 'червоний')
    return cells


rows = []
for i, (name, path, old, new) in enumerate(MUTATIONS, 1):

    finally:

    rows.append('| ' + str(i) + ' | ' + name + ' | ' + ' | '.join(cells) + ' |')
    print(rows[-1], flush=True)
